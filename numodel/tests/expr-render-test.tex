% expr-render-test.tex
% Visual inspection test: how does \textmodel render each l3fp construct?
%
% Purpose: compile and inspect \textmodel output for every operator and
%   function in the l3fp expression grammar, including those not yet
%   display-translated by numodel.  \computemodel verifies that the
%   expressions also evaluate without error.
%
% The model body is wrapped in \fullmodel{<prefix>} and called twice --
% once under syntax=EN (XMILE keywords) and once under syntax=NL
% (CoachTaal keywords) -- so the rendered table appears in both forms
% for side-by-side comparison of the display-translation layer.  The
% prefix-dependent rule bodies use the helper \vv{<name>}, which
% expands to \<current-prefix><name>.
%
% Compile with:  lualatex expr-render-test.tex
%   (run from numodel/ after unpacking, or from build/doc/ after l3build doc)
%
% Model design: U starts at 1.0 and grows 10 % per step, so it stays
%   positive throughout -- safe for ln, sqrt, log, etc.  T steps 0.1 s
%   per iteration; the simulation stops after 3 steps (T >= 0.25).
%
% l3fp notes (cf. interface3.pdf section 30.13.3):
%   - no log(x): l3fp only has ln(x); use ln(x)/ln(10) for base-10
%   - logb(x) is floor(log10(|x|)), NOT log base b
%   - no mod:  use x - trunc(x/y)*y (l3fp roadmap item)
%   - no sinh/cosh/tanh/coth: use explicit (exp(x)-exp(-x))/2 etc.
%   - no e constant: use exp(1)
%   - acsc/asec/acscd/asecd need |x| >= 1
%   - fact(x) requires non-negative integer x

\documentclass{article}
\usepackage[syntax=EN]{numodel}
\usepackage[left=2cm, right=2cm, top=2cm, bottom=2cm]{geometry}
\begin{document}

% \vv{X} -> \<prefix>X (the unexpanded control sequence).  Used inside
% \Mrule / \Mrules / \Mstop bodies so that the rule body resolves to
% the current prefix's variable cs.
\def\vv#1{\expandafter\noexpand\csname\Mprefix#1\endcsname}

% Wrappers that e-expand their body before delegating, so \vv{...} is
% resolved at declaration time.  \mvar and \mruletext do not reference
% prefix variables, so they need no wrapping.
\def\Mrule#1#2{\edef\Mtmp{\noexpand\mrule{#1}{#2}}\Mtmp}
\def\Mrules#1#2{\edef\Mtmp{\noexpand\mrule*{#1}{#2}}\Mtmp}
\def\Mstop#1{\edef\Mtmp{\noexpand\mstop{#1}}\Mtmp}

% The model body, parameterised by prefix.
\newcommand\fullmodel[1]{%

  \newmodelprefix{#1}%
  %
  % --- System / stock variables ---
  \mvar{T}{t}{0}{\s}{3}{system}%
  \mvar{Dt}{dt}{0.1}{\s}{2}{system}%
  \mvar{U}{u}{1.0}{}{4}{stock}%
  %
  % --- Aux variables ---

  \mvar{Atand2v}{v_{48}}{0}{}{4}{aux}%

  \mvar{Falsev}{v_{62}}{0}{}{4}{aux}%

  %
  % --- Rules ---

  \Mrule{T}{\vv{T} + \vv{Dt}}%
  \Mstop{\vv{T} >= 0.25}%
  %
  \mruletext{--- Arithmetic operators ---}%

  \Mrule{Sinv}{sin(\vv{T} * pi)}%

  %
  \mruletext{--- Trigonometry (degrees) ---}%
  \Mrule{Sindv}{sind(\vv{T} * 180)}%
  \Mrule{Cosdv}{cosd(\vv{T} * 180)}%
  \Mrule{Tandv}{tand(\vv{T} * 90)}%
  \Mrule{Asindv}{asind(sind(\vv{T} * 90))}%
  \Mrule{Acosdv}{acosd(cosd(\vv{T} * 90))}%
  \Mrule{Atandv}{atand(\vv{T})}%
  \Mrule{Atand2v}{atand(\vv{T}, 1.0)}%
  \Mrule{Cotdv}{cotd(\vv{T} * 90)}%
  \Mrule{Cscdv}{cscd(\vv{T} * 90)}%
  \Mrule{Secdv}{secd(\vv{T} * 90)}%
  \Mrule{Acotdv}{acotd(\vv{T})}%
  \Mrule{Acotd2v}{acotd(\vv{T}, 1.0)}%
  \Mrule{Acscdv}{acscd(2)}%
  \Mrule{Asecdv}{asecd(2)}%
  %
  \mruletext{--- Hyperbolic (via exp formulas) ---}%
  \Mrule{Sinhv}{(exp(\vv{T}) - exp(-\vv{T})) / 2}%
  \Mrule{Coshv}{(exp(\vv{T}) + exp(-\vv{T})) / 2}%
  \Mrule{Tanhv}{(exp(\vv{T}) - exp(-\vv{T})) / (exp(\vv{T}) + exp(-\vv{T}))}%
  \Mrule{Cothv}{(exp(\vv{T} + 0.5) + exp(-(\vv{T} + 0.5))) / (exp(\vv{T} + 0.5) - exp(-(\vv{T} + 0.5)))}%
  %
  \mruletext{--- Constants ---}%
  \Mrule{Piv}{pi}%
  \Mrule{Ev}{exp(1)}%
  \Mrule{Degv}{deg}%
  \Mrule{Truev}{true}%
  \Mrule{Falsev}{false}%
  %
  \mruletext{--- Random ---}%
  \Mrule{Randv}{rand()}%
  \Mrule{Randint1v}{randint(10)}%
  \Mrule{Randint2v}{randint(5, 15)}%
  %
  % U updated last so all aux rules above see the previous step's value
  \Mrule{U}{\vv{U} * 1.1}%
}
