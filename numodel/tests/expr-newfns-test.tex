%         reciprocals acsc/asec/acot, the degree reciprocals

\documentclass{article}
\usepackage{numodel}

\newmodelprefix{tst}
\mvar{T}{t}{0}{}{2}{system}
\mvar{Dt}{dt}{1}{}{2}{system}
\mvar{X}{x}{1}{}{3}{aux}
\mvar{A}{a}{0}{}{3}{stock}

\mrule{A}{\tstA + atan(\tstX) + cot(\tstX) + csc(\tstX) + sec(\tstX)
          + min(\tstX,\tstDt) + max(\tstX,\tstDt)
          + floor(\tstX) + ceil(\tstX)
          + asind(\tstX) + acosd(\tstX) + atand(\tstX)
          + sind(\tstX) + cosd(\tstX) + tand(\tstX)
          + acsc(\tstX) + asec(\tstX) + acot(\tstX)
          + cscd(\tstX) + cotd(\tstX) + secd(\tstX)
          + acscd(\tstX) + asecd(\tstX) + acotd(\tstX)}
\mrule{T}{\tstT + \tstDt}
\mstop{\tstT >= 2}

\begin{document}
\textmodel

\numodelsetup{syntax=NL}
\textmodel
\end{document}
