prefix=ball
vars:
  ballT: short=T type=system text=t value=0 value_expr=0 unit=\s  sigfigs=2
  ballDt: short=Dt type=system text=dt value=0.05 value_expr=0.05 unit=\s  sigfigs=2
  ballV: short=V type=stock text=v value=0 value_expr=0 unit=\m \per \s  sigfigs=3
  ballY: short=Y type=stock text=y value=100 value_expr=100 unit=\m  sigfigs=3
  ballG: short=G type=constant text=g value=-9.81 value_expr=-9.81 unit=\m \per \s \squared  sigfigs=3
program:
  calc: ballV := \ballV + \ballG * \ballDt 
  calc: ballY := \ballY + \ballV * \ballDt 
  calc: ballT := \ballT + \ballDt 
  stop: \ballY <= 2
stop=\ballY <= 2
